using Microsoft.AspNetCore.Mvc;
using Smart_X_Client.Models;
using Smart_X_Client.Services;
using System.Diagnostics;

namespace Smart_X_Client.Controllers
{
    public class HomeController : Controller
    {
        private readonly ILogger<HomeController> _logger;
        private readonly ApiService _apiService;

        public HomeController(
            ILogger<HomeController> logger,
            ApiService apiService)
        {
            _logger = logger;
            _apiService = apiService;
        }

        public async Task<IActionResult> Index()
        {
            try
            {
                var sensors = await _apiService.GetSensorsAsync();

                return View(sensors);
            }
            catch (Exception ex)
            {
                _logger.LogError(
                    ex,
                    "Unable to connect to Smart-X API.");

                ViewBag.ApiError =
                    "Unable to connect to the Smart-X API.";

                return View(new List<Sensor>());
            }
        }

        public IActionResult Privacy()
        {
            return View();
        }

        [ResponseCache(
            Duration = 0,
            Location = ResponseCacheLocation.None,
            NoStore = true)]
        public IActionResult Error()
        {
            return View(
                new ErrorViewModel
                {
                    RequestId =
                        Activity.Current?.Id ??
                        HttpContext.TraceIdentifier
                });
        }
    }
}
